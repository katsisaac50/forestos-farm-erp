function readCachedUser(): SessionUser | null {
  try {
    const raw = localStorage.getItem('forestos_user');
    if (!raw) return null;

    const value: unknown = JSON.parse(raw);

    if (
      typeof value === 'object' &&
      value !== null &&
      'id' in value &&
      'email' in value &&
      'role' in value &&
      typeof value.id === 'string' &&
      typeof value.email === 'string' &&
      typeof value.role === 'string'
    ) {
      return value as SessionUser;
    }
  } catch {
    // Cached display data is optional and never authenticates a user.
  }

  return null;
}

export class ApiError extends Error {
  constructor(
    public readonly status: number,
    message: string,
    public readonly kind: ApiErrorKind = 'http',
  ) {
    super(message);
    this.name = 'ApiError';
  }
}

function authHeaders(): Record<string, string> {
  if (typeof window === 'undefined') return {};

  const token = localStorage.getItem('forestos_token');
  const orgId =
    localStorage.getItem('forestos_organization_id') ||
    localStorage.getItem('forestos_active_organization');

  const headers: Record<string, string> = {};

  if (token) headers.Authorization = `Bearer ${token}`;
  if (orgId) headers['X-Organization-Id'] = orgId;

  return headers;
}

function parseResponse<T>(text: string): T {
  if (!text) return undefined as T;

  try {
    return JSON.parse(text) as T;
  } catch {
    return text as unknown as T;
  }
}

async function throwResponseError(response: Response): Promise<never> {
  const text = await response.text().catch(() => '');

  if (response.status === 401) {
    throw new ApiError(401, 'Please sign in to continue.');
  }

  if (response.status === 400 && text.includes('X-Organization-Id')) {
    throw new ApiError(
      400,
      'No organization selected. Sign in again to refresh your organization context.',
    );
  }

  throw new ApiError(
    response.status,
    text || `API request failed: ${response.status}`,
  );
}

async function request<T>(
  path: string,
  init: RequestInit = {},
): Promise<T> {
  const headers = {
    ...authHeaders(),
    ...(init.headers as Record<string, string> | undefined),
  };

  let response: Response;

  try {
    response = await fetch(API + path, {
      ...init,
      cache: init.cache ?? 'no-store',
      headers,
    });
  } catch {
    throw new ApiError(
      0,
      'Unable to reach the ForestOS API. Check your connection and try again.',
      'network',
    );
  }

  if (!response.ok) {
    return throwResponseError(response);
  }

  const text = await response.text();
  return parseResponse<T>(text);
}

  let best: { id: string; len: number } | undefined;

import {
  ForbiddenException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { InjectModel } from '@nestjs/mongoose';
import { Model, Types } from 'mongoose';
import { Organization, OrganizationDocument } from './organization.schema';
import {
  OrganizationMembership,
  OrganizationMembershipDocument,
} from './organization-membership.schema';
import { OrganizationContext } from './organization-context';

@Injectable()
export class OrganizationAccessService {
  constructor(
    @InjectModel(Organization.name)
    private readonly organizations: Model<OrganizationDocument>,
    @InjectModel(OrganizationMembership.name)
    private readonly memberships: Model<OrganizationMembershipDocument>,
  ) {}

  /**
   * Resolve a verified active organization context for a user.
   * The organization ID is only a selector: membership and organization status
   * are checked against the database on every call.
   */
  async resolveContext(userId: string, requestedOrganizationId: string): Promise<OrganizationContext> {
    if (!Types.ObjectId.isValid(userId) || !Types.ObjectId.isValid(requestedOrganizationId)) {
      throw new NotFoundException('Organization or membership not found');
    }

    const organization = await this.organizations.findOne({
      _id: new Types.ObjectId(requestedOrganizationId),
      status: 'active',
    }).lean();

    if (!organization) {
      throw new NotFoundException('Organization or membership not found');
    }

    // Membership schema stores both references as strings, so query with the
    // canonical string IDs rather than BSON ObjectId values.
    const membership = await this.memberships.findOne({
      userId: String(userId),
      organizationId: String(requestedOrganizationId),
      status: 'active',
    }).lean();

    if (!membership) {
      throw new ForbiddenException('No active membership for the selected organization');
    }

    return {
      organizationId: String(organization._id),
      userId: String(userId),
      roles: Array.isArray((membership as any).roles)
        ? (membership as any).roles
        : (typeof (membership as any).role === 'string' && (membership as any).role
            ? [(membership as any).role]
            : []),
    };
  }

  async listMyOrganizations(userId: string) {
    if (!Types.ObjectId.isValid(userId)) return [];
    const memberships = await this.memberships.find({
      userId: String(userId),
      status: 'active',
    }).lean();

    const organizationIds = memberships
      .map((m: any) => String(m.organizationId))
      .filter((id: string) => Types.ObjectId.isValid(id));
    if (!organizationIds.length) return [];

    const organizations = await this.organizations.find({
      _id: { $in: organizationIds.map((id: string) => new Types.ObjectId(id)) },
      status: 'active',
    }).lean();

    const byId = new Map(organizations.map((o: any) => [String(o._id), o]));
    return memberships.flatMap((m: any) => {
      const org = byId.get(String(m.organizationId));
      return org ? [{
        organizationId: String(org._id),
        slug: org.slug,
        name: org.name,
        roles: Array.isArray(m.roles)
          ? m.roles
          : (typeof m.role === 'string' && m.role ? [m.role] : []),
      }] : [];
    });
  }
}

import { BadRequestException, ConflictException, Injectable, NotFoundException } from '@nestjs/common';
import { InjectModel } from '@nestjs/mongoose';
import { Model, Types } from 'mongoose';
import { Organization } from './organization.schema';
import { OrganizationMembership } from './organization-membership.schema';
import { OrganizationAudit } from './organization-audit.schema';
import { User } from '../auth/user.schema';

@Injectable()
export class OrganizationsService {
  constructor(
    @InjectModel(Organization.name) private orgs: Model<Organization>,
    @InjectModel(OrganizationMembership.name) private memberships: Model<OrganizationMembership>,
    @InjectModel(OrganizationAudit.name) private audit: Model<OrganizationAudit>,
    @InjectModel(User.name) private users: Model<User>,
  ) {}

  async list() {
    const items = await this.orgs.find({}).sort({ createdAt: -1 }).limit(500).lean();
    return {
      items,
      total: items.length,
      limit: 500,
      note: 'Registry view only; listing does not grant access to organization data.',
    };
  }

  async create(b: any, actor: string) {
    const slug = String(b?.slug || '').trim().toLowerCase();
    const name = String(b?.name || '').trim();
    const reason = String(b?.reason || '').trim();
    const currency = String(b?.baseCurrency || 'UGX').trim().toUpperCase();
    const timezone = String(b?.timezone || 'Africa/Kampala').trim();

    if (!/^[a-z0-9][a-z0-9-]{2,62}$/.test(slug))
      throw new BadRequestException('slug must be 3–63 lowercase letters, numbers, or hyphens');
    if (!name || name.length > 160)
      throw new BadRequestException('name is required (max 160 characters)');
    if (reason.length < 10 || reason.length > 1000)
      throw new BadRequestException('reason must contain 10–1000 characters');
    if (!/^[A-Z]{3}$/.test(currency))
      throw new BadRequestException('baseCurrency must be a 3-letter currency code');

    try {
      const d = await this.orgs.create({
        slug,
        name,
        status: 'active',
        baseCurrency: currency,
        timezone,
        country: String(b?.country || 'Uganda').trim().slice(0, 80),
        createdBy: actor,
        recordVersion: 1,
      });
      await this.audit.create({
        organizationId: String(d._id),
        actor,
        action: 'organization_created',
        reason,
        details: { slug, name, currency, timezone },
      });
      return {
        organization: d.toObject(),
        note: 'No membership is granted automatically; a separate authorized membership grant is required.',
      };
    } catch (e: any) {
      if (e?.code === 11000) throw new ConflictException('Organization slug already exists');
      throw e;
    }
  }

  async addMembership(b: any, actor: string) {
    const organizationId = String(b?.organizationId || '').trim();
    const userIdRaw = String(b?.userId || '').trim().toLowerCase();
    const role = String(b?.role || '');
    const reason = String(b?.reason || '').trim();

    if (!organizationId || !userIdRaw)
      throw new BadRequestException('organizationId and userId are required');

    // NEW: validate the userId is a valid Mongo ObjectId
    if (!Types.ObjectId.isValid(userIdRaw)) {
      throw new BadRequestException(
        'userId must be a valid 24-character Mongo ObjectId (not an email address)',
      );
    }

    if (!['owner', 'admin', 'manager', 'field_worker', 'finance', 'viewer'].includes(role))
      throw new BadRequestException('Unsupported membership role');
    if (reason.length < 10 || reason.length > 1000)
      throw new BadRequestException('reason must contain 10–1000 characters');

    // NEW: verify the organization exists and is active
    const org = await this.orgs.findById(organizationId).lean();
    if (!org) throw new NotFoundException('Organization not found');
    if (org.status !== 'active')
      throw new ConflictException('Organization is suspended');

    // NEW: verify the target user actually exists
    const targetUser = await this.users.findById(userIdRaw).select('_id email active').lean();
    if (!targetUser) {
      throw new NotFoundException('User not found. Create the user account first via /api/auth/users.');
    }
    if (!targetUser.active) {
      throw new ConflictException('User account is deactivated.');
    }

    try {
      const m = await this.memberships.create({
        organizationId,
        userId: userIdRaw,
        role,
        status: 'active',
        addedBy: actor,
      });
      await this.audit.create({
        organizationId,
        actor,
        action: 'membership_added',
        reason,
        details: { userId: userIdRaw, role, membershipId: String(m._id) },
      });
      return m.toObject();
    } catch (e: any) {
      if (e?.code === 11000) throw new ConflictException('Membership already exists');
      throw e;
    }
  }

  async membershipsFor(id: string) {
    if (!(await this.orgs.findById(id).lean()))
      throw new NotFoundException('Organization not found');
    const items = await this.memberships
      .find({ organizationId: id })
      .sort({ createdAt: -1 })
      .limit(1000)
      .lean();
    return { organizationId: id, items, total: items.length };
  }

  async history(id: string, limit = 100) {
    if (!(await this.orgs.findById(id).lean()))
      throw new NotFoundException('Organization not found');
    return this.audit
      .find({ organizationId: id })
      .sort({ createdAt: -1 })
      .limit(Math.max(1, Math.min(500, Math.floor(Number(limit) || 100))))
      .lean();
  }
